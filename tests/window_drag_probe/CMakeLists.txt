cmake_minimum_required(VERSION 3.22)
project(WindowDragProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick DBus Widgets)
find_package(KF6Config REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_path(KWIN_INCLUDE_DIR window.h PATH_SUFFIXES kwin REQUIRED)
find_library(KWIN_LIBRARY kwin REQUIRED)
add_library(dragprobe MODULE probe.cpp)
target_include_directories(dragprobe PRIVATE ${KWIN_INCLUDE_DIR})
target_link_libraries(dragprobe PRIVATE ${KWIN_LIBRARY} Qt6::Quick Qt6::DBus Qt6::Widgets
    KF6::ConfigCore KF6::WindowSystem)
configure_file(qmldir qmldir COPYONLY)
